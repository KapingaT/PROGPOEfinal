# 1. Use the .NET 10 runtime for the final image
FROM mcr.microsoft.com/dotnet/aspnet:10.0 AS base
WORKDIR /app
EXPOSE 8080

# 2. Use the .NET 10 SDK for building
FROM mcr.microsoft.com/dotnet/sdk:10.0 AS build
WORKDIR /src
COPY ["RacedayFINAL.csproj", "./"]
RUN dotnet restore "RacedayFINAL.csproj"
COPY . .
WORKDIR "/src/"
RUN dotnet build "RacedayFINAL.csproj" -c Release -o /app/build

FROM build AS publish
RUN dotnet publish "RacedayFINAL.csproj" -c Release -o /app/publish

FROM base AS final
WORKDIR /app
COPY --from=publish /app/publish .
ENTRYPOINT ["dotnet", "RacedayFINAL.dll"]
